```ts
// 浏览器 TTS 受限,只能用音高/语速区分角色;火山云 TTS 升级后换真实音色(见计划文末)
export interface VoiceParam { pitch: number; rate: number }

// personaId → 参数。pitch:0(低沉)~2(尖),rate:0.1~10(常用 0.8~1.2)
export const VOICE_PARAMS: Record<string, VoiceParam> = {
  laopaoer: { pitch: 0.8, rate: 1.05 }, // 老炮儿:低沉、略快、冲
  coach: { pitch: 1.0, rate: 0.95 },    // 林教练:平稳、慢条斯理
  netizen: { pitch: 1.25, rate: 1.18 }, // 吃瓜阿强:偏高、快、跳脱
  player: { pitch: 0.9, rate: 1.0 },    // 老黑:沉稳
}

export const DEFAULT_VOICE: VoiceParam = { pitch: 1.0, rate: 1.0 }

export function voiceParam(personaId: string): VoiceParam {
  return VOICE_PARAMS[personaId] ?? DEFAULT_VOICE
}
```

- [ ] **Step 2: SpeechSynthesis 封装**

Create `web/src/voice/browser-tts.ts`:
```ts
import { voiceParam } from './voice-config.js'

// 浏览器 TTS。语音列表异步加载,挑一个中文嗓子;角色用音高/语速区分。
let zhVoice: SpeechSynthesisVoice | null = null
let muted = false

function pickZhVoice(): SpeechSynthesisVoice | null {
  if (typeof speechSynthesis === 'undefined') return null
  const voices = speechSynthesis.getVoices()
  return voices.find(v => v.lang?.toLowerCase().startsWith('zh')) ?? null
}

// 语音列表可能首调用为空,监听 voiceschanged 补齐
if (typeof speechSynthesis !== 'undefined') {
  zhVoice = pickZhVoice()
  speechSynthesis.onvoiceschanged = () => { zhVoice = pickZhVoice() }
}

export function isTtsSupported(): boolean {
  return typeof window !== 'undefined' && 'speechSynthesis' in window
}

export function setMuted(m: boolean): void {
  muted = m
  if (m && typeof speechSynthesis !== 'undefined') speechSynthesis.cancel()
}

export function isMuted(): boolean { return muted }

/** 念一句;静音或不支持时静默跳过。多句自动排队(SpeechSynthesis 内部队列)。 */
export function speak(personaId: string, text: string): void {
  if (muted || !isTtsSupported() || !text) return
  const u = new SpeechSynthesisUtterance(text)
  const { pitch, rate } = voiceParam(personaId)
  u.lang = 'zh-CN'
  u.pitch = pitch
  u.rate = rate
  if (!zhVoice) zhVoice = pickZhVoice()
  if (zhVoice) u.voice = zhVoice
  speechSynthesis.speak(u)
}

/** 立刻停掉当前与排队的播报(如重开一局) */
export function cancelSpeech(): void {
  if (typeof speechSynthesis !== 'undefined') speechSynthesis.cancel()
}
```

## Task 2: 中文语音输入 hook

**Files:** Create `web/src/voice/use-speech-input.ts`

- [ ] **Step 1: SpeechRecognition hook**

Create `web/src/voice/use-speech-input.ts`:
```ts
import { useRef, useState, useCallback } from 'react'

// webkitSpeechRecognition 不在标准 TS lib,用最小 any 取用(不加 @types)
function getRecognitionCtor(): any {
  if (typeof window === 'undefined') return null
  return (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition || null
}

export function isSttSupported(): boolean {
  return getRecognitionCtor() !== null
}

/**
 * 一次性中文听写。调用 start() 开始听,识别到结果回调 onResult(text) 并自动停止。
 * 返回 listening 状态用于 UI(麦克风高亮)。
 */
export function useSpeechInput(onResult: (text: string) => void) {
  const [listening, setListening] = useState(false)
  const recRef = useRef<any>(null)

  const start = useCallback(() => {
    const Ctor = getRecognitionCtor()
    if (!Ctor || listening) return
    const rec = new Ctor()
    recRef.current = rec
    rec.lang = 'zh-CN'
    rec.interimResults = false
    rec.maxAlternatives = 1
    rec.onresult = (e: any) => {
      const text = e.results?.[0]?.[0]?.transcript ?? ''
      if (text) onResult(text)
    }
    rec.onend = () => setListening(false)
    rec.onerror = () => setListening(false)
    setListening(true)
    rec.start()
  }, [listening, onResult])

  const stop = useCallback(() => {
    recRef.current?.stop?.()
    setListening(false)
  }, [])

  return { listening, start, stop, supported: isSttSupported() }
}
```

## 已知简化(ponytail)

- 浏览器 TTS 音色有限,只能音高/语速区分角色,不如云端自然(升级见下)。
- STT 为一次性听写(按一次说一句),非持续聆听;够用且省事。
- 浏览器兼容性:Chrome/Edge 最佳;Safari 的 SpeechRecognition 支持有限;Firefox STT 基本缺失——已做特性检测降级。
- 无前端测试框架,语音逻辑靠 typecheck+build+手动验收(web 现状如此,不为此引入 vitest)。
