export function pickSpeakers(
  stim: Stimulus, candidates: Persona[],
  recentMap: Map<string, number>, rng: () => number,
  threshold: number, jitterWidth: number, maxSpeakers: number,
): PersonaId[] {
  const scored = candidates.map(p => {
    const base = stim.tension
    const jitter = (rng() - 0.5) * jitterWidth
    const sinceLast = recentMap.get(p.id)
    const penalty = sinceLast != null && sinceLast < RECENT_PENALTY_WINDOW_S
      ? (1 - sinceLast / RECENT_PENALTY_WINDOW_S) * 0.5
      : 0
    return { id: p.id as PersonaId, score: base + jitter - penalty }
  })
  return scored
    .filter(s => s.score >= threshold)
    .sort((a, b) => b.score - a.score)
    .slice(0, maxSpeakers)
    .map(s => s.id)
}
```

/** idle 闲话是否该触发:仅在有进行中对局、未暂停、距上一条弹幕沉默够久、且掷骰命中时。 */
export function idleShouldFire(a: {
  hasGame: boolean; paused: boolean; silentMs: number; silenceThresholdMs: number; roll: number; prob: number
}): boolean {
  if (!a.hasGame || a.paused) return false
  if (a.silentMs < a.silenceThresholdMs) return false
  return a.roll < a.prob
}
```
