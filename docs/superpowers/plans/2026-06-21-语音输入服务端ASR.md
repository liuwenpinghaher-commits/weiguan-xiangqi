# 语音输入(服务端 ASR)Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把用户语音输入从浏览器 `webkitSpeechRecognition`(依赖 Google,大陆不可达)换成服务端 ASR:前端录音→编码 WAV→`POST /asr`→后端调阿里云 `qwen3-asr-flash`→返回文字→直接发送。

**Architecture:** 后端新增 `AsrProvider` 抽象(`DashscopeAsr` 走阿里云 OpenAI 兼容 `chat/completions` + `input_audio`;`FakeAsr` 供测试)+ `POST /asr` 路由。前端用 MediaRecorder 录音 → `decodeAudioData` 解码 → 降采样单声道 16k → 编码 16-bit PCM WAV → POST。复用现有 DASHSCOPE 凭据(与 LLM 同 `compatible-mode/v1` baseURL)。

**Tech Stack:** TypeScript(ESM,`.js` 后缀)、Fastify 4、`openai` SDK(已用于 DashscopeLLM/Embedding)、阿里云百炼 `qwen3-asr-flash`、浏览器 Web Audio API(getUserMedia/MediaRecorder/AudioContext)、vitest(server node 测;web 纯逻辑测 + 组件 typecheck/build)。

参考 spec:[2026-06-21-语音输入服务端ASR-design.md](../specs/2026-06-21-语音输入服务端ASR-design.md)

## 已核实的阿里云 ASR 接口(据官方文档)
- 端点(OpenAI 兼容,北京):`POST https://dashscope.aliyuncs.com/compatible-mode/v1/chat/completions`(= LLM 同一 baseURL)。
- 请求:`model:'qwen3-asr-flash'`,`messages:[{role:'user',content:[{type:'input_audio',input_audio:{data:'data:audio/wav;base64,<b64>'}}]}]`。
- 响应:`choices[0].message.content`(字符串=转写文字)。
- 音频:文档明确 **WAV(audio/wav)/ MP3(audio/mpeg)**;base64 后 ≤10MB。**webm/opus 未文档化** → 前端必须产出 WAV。
- 文档:https://help.aliyun.com/zh/model-studio/qwen-asr-api-reference

```ts
export interface AsrProvider {
  /** 整段音频转写为文字;失败抛错;无法识别返回空串 */
  transcribe(audio: Buffer, mimeType: string): Promise<string>
}

type Responder = string | ((audio: Buffer, mimeType: string) => string)

/** 测试用:返回固定文字或由函数决定,不打网络。 */
export class FakeAsr implements AsrProvider {
  constructor(private responder: Responder = '') {}
  async transcribe(audio: Buffer, mimeType: string): Promise<string> {
    return typeof this.responder === 'function' ? this.responder(audio, mimeType) : this.responder
  }
}
```

```ts
import OpenAI from 'openai'
import type { AsrProvider } from './asr.js'

/** 阿里云百炼 qwen3-asr-flash(OpenAI 兼容 chat/completions + input_audio)。
 *  client 可注入以便测试不打网络;复用与 LLM 相同的 compatible-mode/v1 baseURL。 */
export class DashscopeAsr implements AsrProvider {
  private client: { chat: { completions: { create: (b: any) => Promise<any> } } }

  constructor(
    private model = process.env.ASR_MODEL ?? 'qwen3-asr-flash',
    apiKey?: string,
    baseURL?: string,
    client?: { chat: { completions: { create: (b: any) => Promise<any> } } },
  ) {
    if (client) {
      this.client = client
    } else {
      if (!apiKey || !baseURL) throw new Error('缺少 DASHSCOPE_API_KEY / DASHSCOPE_BASE_URL(见 .env)')
      this.client = new OpenAI({ apiKey, baseURL }) as any
    }
  }

  async transcribe(audio: Buffer, mimeType: string): Promise<string> {
    const dataUri = `data:${mimeType};base64,${audio.toString('base64')}`
    const body: any = {
      model: this.model,
      messages: [{ role: 'user', content: [{ type: 'input_audio', input_audio: { data: dataUri } }] }],
    }
    const resp: any = await this.client.chat.completions.create(body)
    const content = resp.choices?.[0]?.message?.content
    return typeof content === 'string' ? content.trim() : ''
  }
}
```

```ts
/** Float32 单声道样本 → 16-bit PCM 单声道 WAV(ArrayBuffer)。 */
export function encodeWav(samples: Float32Array, sampleRate: number): ArrayBuffer {
  const n = samples.length
  const buffer = new ArrayBuffer(44 + n * 2)
  const view = new DataView(buffer)
  const w = (off: number, s: string) => { for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i)) }
  w(0, 'RIFF'); view.setUint32(4, 36 + n * 2, true); w(8, 'WAVE'); w(12, 'fmt ')
  view.setUint32(16, 16, true); view.setUint16(20, 1, true); view.setUint16(22, 1, true)
  view.setUint32(24, sampleRate, true); view.setUint32(28, sampleRate * 2, true)
  view.setUint16(32, 2, true); view.setUint16(34, 16, true); w(36, 'data'); view.setUint32(40, n * 2, true)
  let off = 44
  for (let i = 0; i < n; i++) {
    const s = Math.max(-1, Math.min(1, samples[i]))
    view.setInt16(off, s < 0 ? s * 0x8000 : s * 0x7fff, true)
    off += 2
  }
  return buffer
}

/** 线性抽取降采样到单声道目标率;目标率≥源率时原样返回。 */
export function downsampleMono(samples: Float32Array, fromRate: number, toRate: number): Float32Array {
  if (toRate >= fromRate) return samples
  const ratio = fromRate / toRate
  const outLen = Math.floor(samples.length / ratio)
  const out = new Float32Array(outLen)
  for (let i = 0; i < outLen; i++) out[i] = samples[Math.floor(i * ratio)]
  return out
}
```

```ts
import { useRef, useState, useCallback } from 'react'
import { ASR_URL } from '../api/config.js'
import { encodeWav, downsampleMono } from './wav.js'

const TARGET_RATE = 16000

export function isSttSupported(): boolean {
  return typeof window !== 'undefined'
    && !!navigator.mediaDevices?.getUserMedia
    && typeof window.MediaRecorder !== 'undefined'
    && typeof window.AudioContext !== 'undefined'
}

/**
 * 点击开始录音(start),再点停止(stop)→ 录到的音频编码为 16k 单声道 WAV,
 * POST 给 /asr,识别出的文字回调 onResult(text)。
 * error 取值:'not-allowed'(麦克风权限)/'asr-failed'/'no-speech'/'unsupported'。
 */
export function useSpeechInput(onResult: (text: string) => void) {
  const [listening, setListening] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const recRef = useRef<MediaRecorder | null>(null)
  const chunksRef = useRef<Blob[]>([])
  const streamRef = useRef<MediaStream | null>(null)

  const start = useCallback(async () => {
    if (listening) return
    setError(null)
    if (!isSttSupported()) { setError('unsupported'); return }
    let stream: MediaStream
    try {
      stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    } catch {
      setError('not-allowed'); return
    }
    streamRef.current = stream
    chunksRef.current = []
    const rec = new MediaRecorder(stream)
    recRef.current = rec
    rec.ondataavailable = (e) => { if (e.data.size > 0) chunksRef.current.push(e.data) }
    rec.onstop = async () => {
      stream.getTracks().forEach((t) => t.stop())
      try {
        const blob = new Blob(chunksRef.current, { type: rec.mimeType || 'audio/webm' })
        const wav = await blobToWav(blob)
        const res = await fetch(`${ASR_URL}?mime=audio/wav`, {
          method: 'POST', headers: { 'Content-Type': 'application/octet-stream' }, body: wav,
        })
        if (!res.ok) { setError(res.status === 503 ? 'asr-unconfigured' : 'asr-failed'); return }
        const { text } = (await res.json()) as { text: string }
        if (text && text.trim()) onResult(text.trim())
        else setError('no-speech')
      } catch {
        setError('asr-failed')
      } finally {
        setListening(false)
      }
    }
    rec.start()
    setListening(true)
  }, [listening, onResult])

  const stop = useCallback(() => { recRef.current?.stop() }, [])

  return { listening, start, stop, supported: isSttSupported(), error }
}

/** webm/opus Blob → 16k 单声道 16-bit WAV Blob(用 AudioContext 解码) */
async function blobToWav(blob: Blob): Promise<Blob> {
  const arrayBuf = await blob.arrayBuffer()
  const ctx = new AudioContext()
  try {
    const audioBuf = await ctx.decodeAudioData(arrayBuf)
    // 混为单声道
    const ch = audioBuf.numberOfChannels
    const len = audioBuf.length
    const mono = new Float32Array(len)
    for (let c = 0; c < ch; c++) {
      const data = audioBuf.getChannelData(c)
      for (let i = 0; i < len; i++) mono[i] += data[i] / ch
    }
    const down = downsampleMono(mono, audioBuf.sampleRate, TARGET_RATE)
    return new Blob([encodeWav(down, TARGET_RATE)], { type: 'audio/wav' })
  } finally {
    void ctx.close()
  }
}
```
