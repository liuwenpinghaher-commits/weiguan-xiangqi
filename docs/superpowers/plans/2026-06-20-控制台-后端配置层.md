# 控制台 · 后端配置层(RuntimeConfig)Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把散落在代码常量/`.env` 的配置收敛到"代码默认值 ⊕ SQLite 覆盖"的 `RuntimeConfig` 层,提供 `/admin/*` 路由做增删改查,改动即时生效(LLM 经 `MutableLLM` 热重载),消费端改为读 cfg。

**Architecture:** 新增 `config/`(defaults + 校验 + RuntimeConfig)与 `store/settings-repo`(`app_settings`+`persona_config` 两表)。`MutableLLM` 包装可替换内层 provider,改 llm/tts 配置时 `rebuild`。`/admin/*` 是一个 Fastify 插件,依赖注入 `cfg`/`mem`/各 repo。消费端(chat-engine/speaker-select/game-session/ws-server/tts)从 cfg 读值。记忆 CRUD 直接走已交付的 `MemoryStore`。本计划是后端;前端 ConsolePage 是后续 Plan B。

**Tech Stack:** TypeScript(ESM,`.js` import 后缀)、Fastify 4(`app.inject` 集成测试)、better-sqlite3(同步)、vitest。

import type { DB } from './db.js'

export interface PersonaConfig {
  id: string; name: string; isPlayer: boolean; skill: string
  background: string; voiceStyle: string; leaning: string; ttsVoice: string; enabled: boolean
}

/** app_settings(段覆盖 JSON)+ persona_config(逐角色覆盖)的 SQLite 访问层。 */
export class SettingsRepo {
  constructor(private db: DB) {}

  getSection(section: string): Record<string, unknown> | null {
    const row = this.db.prepare('SELECT json FROM app_settings WHERE section=?').get(section) as { json: string } | undefined
    return row ? JSON.parse(row.json) : null
  }

  setSection(section: string, json: Record<string, unknown>, updatedAt: string): void {
    this.db.prepare(
      `INSERT INTO app_settings (section, json, updated_at) VALUES (?,?,?)
       ON CONFLICT(section) DO UPDATE SET json=excluded.json, updated_at=excluded.updated_at`,
    ).run(section, JSON.stringify(json), updatedAt)
  }

  personaConfigEmpty(): boolean {
    const row = this.db.prepare('SELECT COUNT(*) AS n FROM persona_config').get() as { n: number }
    return row.n === 0
  }

  seedPersonas(personas: PersonaConfig[], updatedAt: string): void {
    const stmt = this.db.prepare(
      `INSERT OR IGNORE INTO persona_config
       (id, name, is_player, skill, background, voice_style, leaning, tts_voice, enabled, updated_at)
       VALUES (?,?,?,?,?,?,?,?,?,?)`,
    )
    const tx = this.db.transaction((rows: PersonaConfig[]) => {
      for (const p of rows) stmt.run(p.id, p.name, p.isPlayer ? 1 : 0, p.skill, p.background, p.voiceStyle, p.leaning, p.ttsVoice, p.enabled ? 1 : 0, updatedAt)
    })
    tx(personas)
  }

  getAllPersonas(): PersonaConfig[] {
    const rows = this.db.prepare('SELECT * FROM persona_config ORDER BY id').all() as any[]
    return rows.map(r => ({
      id: r.id, name: r.name, isPlayer: !!r.is_player, skill: r.skill,
      background: r.background, voiceStyle: r.voice_style, leaning: r.leaning,
      ttsVoice: r.tts_voice, enabled: !!r.enabled,
    }))
  }

  updatePersona(id: string, patch: Partial<Omit<PersonaConfig, 'id'>>, updatedAt: string): void {
    const cols: string[] = []; const vals: any[] = []
    const map: Record<string, string> = { name: 'name', isPlayer: 'is_player', skill: 'skill', background: 'background', voiceStyle: 'voice_style', leaning: 'leaning', ttsVoice: 'tts_voice', enabled: 'enabled' }
    for (const [k, col] of Object.entries(map)) {
      if (k in patch) {
        const v = (patch as any)[k]
        cols.push(`${col}=?`); vals.push(typeof v === 'boolean' ? (v ? 1 : 0) : v)
      }
    }
    if (!cols.length) return
    cols.push('updated_at=?'); vals.push(updatedAt)
    vals.push(id)
    this.db.prepare(`UPDATE persona_config SET ${cols.join(', ')} WHERE id=?`).run(...vals)
  }

  resetPersona(id: string): void {
    this.db.prepare('DELETE FROM persona_config WHERE id=?').run(id)
  }
}
```

import type { ConfigSection } from './defaults.js'

const clamp = (n: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, n))

/** 每段允许字段 → 清洗函数。返回 undefined 表示丢弃该字段。 */
type Cleaner = (v: unknown) => unknown
const prob: Cleaner = (v) => (typeof v === 'number' && isFinite(v) ? clamp(v, 0, 1) : undefined)
const intMin0: Cleaner = (v) => (typeof v === 'number' && isFinite(v) ? Math.max(0, Math.round(v)) : undefined)
const str: Cleaner = (v) => (typeof v === 'string' && v.trim() !== '' ? v : undefined)
const intRange = (lo: number, hi: number): Cleaner => (v) => (typeof v === 'number' && isFinite(v) ? clamp(Math.round(v), lo, hi) : undefined)
const numRange = (lo: number, hi: number): Cleaner => (v) => (typeof v === 'number' && isFinite(v) ? clamp(v, lo, hi) : undefined)

function modeCleaner(v: unknown): unknown {
  if (typeof v !== 'object' || v === null) return undefined
  const m = v as any; const out: any = {}
  if (typeof m.maxDropCp === 'number') out.maxDropCp = clamp(Math.round(m.maxDropCp), 0, 2000)
  if (m.targetBand && typeof m.targetBand === 'object') {
    const tb: any = {}
    if (typeof m.targetBand.low === 'number') tb.low = clamp(Math.round(m.targetBand.low), -2000, 2000)
    if (typeof m.targetBand.high === 'number') tb.high = clamp(Math.round(m.targetBand.high), -2000, 2000)
    if ('low' in tb || 'high' in tb) out.targetBand = tb
  }
  if (typeof m.strength === 'number') out.strength = clamp(m.strength, 0, 1)
  return Object.keys(out).length ? out : undefined
}

const SCHEMA: Record<ConfigSection, Record<string, Cleaner>> = {
  chat: {
    minSilenceMs: intMin0, maxPerMin: intMin0, speakThreshold: prob, jitterWidth: numRange(0, 2),
    idleProb: prob, idleIntervalMs: intMin0, reactProb: prob, lenShortP: prob, lenLongP: prob,
  },
  director: { coach: modeCleaner, humanlike: modeCleaner },
  engine: { pikafishPath: (v) => (typeof v === 'string' ? v : undefined), multipv: intRange(1, 10), movetime: intRange(50, 5000) },
  diagnose: { depth: intRange(1, 30) },
  llm: { apiKey: str, model: str, baseURL: str },
  tts: { endpoint: str, model: str },
}

/** 清洗一段的部分覆盖:钳制数值、丢弃未知字段与非法值。未知 section 抛错。 */
export function validateSection(section: ConfigSection, patch: Record<string, unknown>): Record<string, unknown> {
  const cleaners = SCHEMA[section]
  if (!cleaners) throw new Error(`unknown config section: ${section}`)
  const out: Record<string, unknown> = {}
  for (const [k, clean] of Object.entries(cleaners)) {
    if (k in patch) {
      const v = clean(patch[k])
      if (v !== undefined) out[k] = v
    }
  }
  return out
}
```

`server/src/config/runtime-config.ts`:
```ts
import type { SettingsRepo } from '../store/settings-repo.js'
import { DEFAULTS, type ChatCfg, type DirectorCfg, type EngineCfg, type DiagnoseCfg, type LlmCfg, type TtsCfg, type ConfigSection } from './defaults.js'
import { validateSection } from './validate.js'

/** 代码默认值 ⊕ 数据库覆盖 的唯一读取入口。每次读都合并,保证即时生效。 */
export class RuntimeConfig {
  /** 由 ws-server 设置:llm/tts 段变更时重建 MutableLLM 内层 */
  onLlmChange?: () => void

  constructor(private repo: SettingsRepo, private now: () => string = () => new Date().toISOString()) {}

  private merged<T>(section: ConfigSection): T {
    const base = DEFAULTS[section] as any
    const ov = this.repo.getSection(section) ?? {}
    return { ...base, ...ov } as T
  }

  getChat(): ChatCfg { return this.merged<ChatCfg>('chat') }
  getEngineParams(): EngineCfg { return this.merged<EngineCfg>('engine') }
  getDiagnose(): DiagnoseCfg { return this.merged<DiagnoseCfg>('diagnose') }
  getTts(): TtsCfg { return this.merged<TtsCfg>('tts') }

  /** director 需按 mode 深合并 */
  getDirectorDefaults(): DirectorCfg {
    const ov = (this.repo.getSection('director') ?? {}) as Partial<DirectorCfg>
    const merge = (m: 'coach' | 'humanlike') => ({
      ...DEFAULTS.director[m], ...(ov[m] ?? {}),
      targetBand: { ...DEFAULTS.director[m].targetBand, ...((ov[m] as any)?.targetBand ?? {}) },
    })
    return { coach: merge('coach'), humanlike: merge('humanlike') }
  }

  /** llm.apiKey/baseURL 缺省回退 env(库无值时的首次引导) */
  getLlm(): LlmCfg {
    const m = this.merged<LlmCfg>('llm')
    return {
      apiKey: m.apiKey ?? process.env.DASHSCOPE_API_KEY ?? null,
      model: m.model,
      baseURL: m.baseURL ?? process.env.DASHSCOPE_BASE_URL ?? null,
    }
  }

  /** 写一段覆盖(校验后)。llm/tts 触发热重载钩子。 */
  updateSection(section: ConfigSection, patch: Record<string, unknown>): void {

/** key 脱敏:保留前 3 后 4,中间 ****;过短全打码;空值返回空串。 */
export function maskKey(key: string | null | undefined): string {
  if (!key) return ''
  if (key.length <= 7) return '****'
  return `${key.slice(0, 3)}****${key.slice(-4)}`

export function pickSpeakers(
  stim: Stimulus, candidates: Persona[],
  recentMap: Map<string, number>, rng: () => number,
  threshold: number, jitterWidth: number,
): PersonaId[] {
  const scored = candidates.map(p => {
    const base = stim.tension
    const jitter = (rng() - 0.5) * jitterWidth
    const sinceLast = recentMap.get(p.id)
    const penalty = sinceLast != null && sinceLast < RECENT_PENALTY_WINDOW_S
      ? (1 - sinceLast / RECENT_PENALTY_WINDOW_S) * 0.5
      : 0
    return { id: p.id as PersonaId, score: base + jitter - penalty }
  })
  return scored
    .filter(s => s.score >= threshold)
    .sort((a, b) => b.score - a.score)
    .slice(0, MAX_SPEAKERS)
    .map(s => s.id)
}
```
